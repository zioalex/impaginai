# Floors, not exact pins.
#
# The first version of this file pinned whatever happened to be installed on the
# machine it was written on, which was months behind, and produced 64 Dependabot
# alerts across pillow and pypdf. Both libraries parse untrusted binary input —
# photographs and PDFs — so they are exactly the two that must not lag.
#
# These are the oldest versions supported, with a ceiling on the majors that
# have already changed under this project. requirements.lock.txt is the exact
# set CI tested, and what RUNNING.md tells people to install; AGENTS.md says
# how to regenerate it.
#
# Check for new advisories at any time with:
#
#     pip install pip-audit && pip-audit -r requirements.txt
#
# Versions in the comments are what this was last verified against.

# --- rendering and print chain (Phase 0) ---
playwright>=1.63.0,<2            # verified 1.63.0
pillow>=13.0.0,<14                # verified 13.0.0 — below this, 39 known CVEs
pillow-heif>=1.1.0            # HEIC/HEIF: the default format on every iPhone
                              # since 2017. Pillow cannot read it alone.
pypdf>=6.21.4,<8                 # verified 7.0.0  — below this, 25 known CVEs

# --- schema, config, planning (Phase 1) ---
pydantic>=2.13.5              # verified 2.13.5
PyYAML>=6.0.3                 # verified 6.0.3
httpx>=0.28.1                 # verified 0.28.1 — OpenAI-compatible endpoints

# Only needed for --provider anthropic, which is the default. Drop it if you
# point --base-url at an OpenAI-compatible endpoint instead.
anthropic>=0.69.0,<3

# --- web editor (Phase 2) ---
fastapi>=0.162.2              # verified 0.162.2
uvicorn>=0.54.0               # verified 0.54.0

# --- optional ---
# Offline place names from GPS live in requirements-geo.txt, not here.
# reverse-geocoder has no wheel for some platforms and builds from source;
# where that build fails, pip abandons the whole resolution and NOTHING is
# installed — so listing an optional dependency here made every install as
# fragile as its least portable member. Install it separately:
#
#     pip install -r requirements-geo.lock.txt
#
# Without it the album simply has no place names.

# Writing EXIF, needed only by the demo photo generator.
piexif>=1.1.3
